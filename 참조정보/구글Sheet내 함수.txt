function sortByWinRate() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("사용자목록"); // 시트 이름 변경 가능
  var range = sheet.getDataRange();
  var values = range.getValues(); // 셀 값 가져오기
  var formulas = range.getFormulas(); // 셀의 함수 가져오기

  var header = values[0]; // 헤더 저장
  var data = values.slice(1); // 헤더 제외한 데이터

  var winRateIndex = 10; // K열 (승률) 인덱스 (0부터 시작)
  var totalGamesIndex = 7; // H열 (총 경기수) 인덱스
  var totalWinsIndex = 8; // I열 (총 승리수) 인덱스
  var minGames = 100; // 최소 경기 수

  // H열이 함수인 경우에도 값을 가져와서 비교
  var dataWithValues = data.map((row, i) => {
    return {
      index: i, // 원래 인덱스를 저장하여 나중에 순서 복원
      values: row, // 원래 데이터 저장
      totalGames: sheet.getRange(i + 2, totalGamesIndex + 1).getValue(), // H열의 실제 값 가져오기
      winRate: row[winRateIndex], // K열 (승률)
      totalWins: row[totalWinsIndex] // I열 (총 승리수)
    };
  });

  // H열이 100 이상이면 승률(K열) 기준 정렬, 미만이면 총 승리수(I열) 기준 정렬
  dataWithValues.sort((a, b) => {
    if (a.totalGames >= minGames && b.totalGames >= minGames) {
      return b.winRate - a.winRate; // 승률 기준 내림차순 정렬
    } else if (a.totalGames < minGames && b.totalGames < minGames) {
      return b.totalWins - a.totalWins; // 총 승리수 기준 내림차순 정렬
    } else {
      return a.totalGames >= minGames ? -1 : 1; // 100 이상인 데이터를 우선 정렬
    }
  });

  // 정렬된 데이터만 추출하여 업데이트 (수식이 있는 경우 수식 복원)
  var sortedValues = dataWithValues.map(rowData => rowData.values);

  // 기존 데이터 삭제 후, 정렬된 데이터 삽입
  sheet.getRange(2, 1, data.length, data[0].length).clearContent();
  sheet.getRange(2, 1, sortedValues.length, sortedValues[0].length).setValues(sortedValues);

  // 모든 수식 복원
  for (var r = 1; r < values.length; r++) { // 1부터 시작 (헤더 제외)
    for (var c = 0; c < values[0].length; c++) {
      if (formulas[r][c]) { // 해당 셀이 함수라면
        sheet.getRange(r + 1, c + 1).setFormula(formulas[r][c]); // 원래 수식 복원
      }
    }
  }
}


function inputPointHist(userId, dPlayDate, point, gameId){
  let ss= SpreadsheetApp.getActiveSpreadsheet();

  let datasheet = ss.getSheetByName("포인트변동");

  datasheet.insertRowAfter(1);

  //순서
  datasheet.getRange(2,1).setFormula("=Row()-1");

  //경기날짜
  datasheet.getRange(2,2).setValue(dPlayDate);

  //ID
  datasheet.getRange(2,3).setValue(userId);

  //포인트
  datasheet.getRange(2,4).setValue(point);

  //시즌
  datasheet.getRange(2,5).setValue(getCurrentSeason());

  //게임ID
  datasheet.getRange(2,6).setValue(gameId);

}

function updateTeamStats(teamMembers, result, statsSheetName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const statsSheet = ss.getSheetByName(statsSheetName);

  if (!statsSheet) {
    throw new Error('통계 시트를 찾을 수 없습니다: ' + statsSheetName);
  }

  const currentSeason = getCurrentSeason();
  const lastRow = statsSheet.getLastRow();
  const seasonColumnIndex = 1; // A열
  const duoColumnIndex = 2; // B열
  const winsColumnIndex = 3; // C열
  const lossesColumnIndex = 4; // D열

  // 시트에서 현재 시즌 데이터만 효율적으로 검색하기 위해 맵을 사용합니다.
  const duoStatsMap = new Map();
  if (lastRow > 1) {
    const dataRange = statsSheet.getRange(2, 1, lastRow - 1, statsSheet.getLastColumn());
    const values = dataRange.getValues();
    for (const row of values) {
      if (row[seasonColumnIndex - 1] === currentSeason) {
        duoStatsMap.set(row[duoColumnIndex - 1], {
          wins: row[winsColumnIndex - 1],
          losses: row[lossesColumnIndex - 1],
          rowIndex: values.indexOf(row) + 2
        });
      }
    }
  }

  const updates = [];
  const newRows = [];

  // 팀원들로 가능한 모든 2인 조합을 만듭니다.
  for (let i = 0; i < teamMembers.length; i++) {
    for (let j = i + 1; j < teamMembers.length; j++) {
      const player1 = teamMembers[i];
      const player2 = teamMembers[j];
      const combination = [player1, player2].sort().join('&');
      
      const existingStat = duoStatsMap.get(combination);

      if (existingStat) {
        // 기존 조합에 대한 승/패 횟수를 업데이트합니다.
        const updatedWins = existingStat.wins + (result === 1 ? 1 : 0);
        const updatedLosses = existingStat.losses + (result === 0 ? 1 : 0);
        updates.push({
          rowIndex: existingStat.rowIndex,
          values: [updatedWins, updatedLosses]
        });
      } else {
        // 새로운 조합이면 새 행을 추가할 준비를 합니다.
        const winValue = result === 1 ? 1 : 0;
        const lossValue = result === 0 ? 1 : 0;
        newRows.push([currentSeason, combination, winValue, lossValue, '']);
      }
    }
  }

  // 1. 기존 행 업데이트
  // 이 부분은 기존 방식과 큰 차이가 없지만, 업데이트할 행을 정확히 찾습니다.
  for (const update of updates) {
    statsSheet.getRange(update.rowIndex, winsColumnIndex, 1, 2).setValues([update.values]);
  }

  // 2. 새 행 추가 및 승률 수식 적용
  if (newRows.length > 0) {
    statsSheet.getRange(statsSheet.getLastRow() + 1, 1, newRows.length, newRows[0].length).setValues(newRows);
    
    // 새로 추가된 행들에 대한 승률 수식 일괄 적용
    const newStartRow = statsSheet.getLastRow() - newRows.length + 1;
    for (let i = 0; i < newRows.length; i++) {
      const rowIndex = newStartRow + i;
      const winRateFormula = `=IFERROR(TEXT(C${rowIndex}/(C${rowIndex}+D${rowIndex}), "0.0%"), "N/A")`;
      statsSheet.getRange(rowIndex, 5).setFormula(winRateFormula);
    }
  }
}

// 4명의 플레이어로부터 모든 듀오 조합을 생성하는 헬퍼 함수 (기존 함수 유지)
function generateDuos(players) {
  const duos = [];
  if (players.length !== 4) return duos;

  for (let i = 0; i < players.length - 1; i++) {
    for (let j = i + 1; j < players.length; j++) {
      const p1 = players[i];
      const p2 = players[j];
      const sortedDuo = [p1, p2].sort();
      duos.push(sortedDuo);
    }
  }
  return duos;
}


function sortStatsSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('듀오전적');  
  
  if (!sheet) {
    SpreadsheetApp.getUi().alert('\'듀오전적\' 시트를 찾을 수 없습니다.');
    return;
  }

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    // 헤더만 있거나 데이터가 없는 경우 정렬할 필요가 없음
    return;
  }

  // 헤더를 제외한 모든 데이터를 가져옵니다.
  const dataRange = sheet.getRange(2, 1, lastRow - 1, 4);
  const data = dataRange.getValues();

  // 정렬 로직
  data.sort((a, b) => {
    const seasonA = a[0]; // 시즌 정보 (A열)
    const seasonB = b[0]; // 시즌 정보 (A열)
    const winsA = a[2]; // 승리수 (C열)
    const winsB = b[2]; // 승리수 (C열)
    const lossesA = a[3]; // 패배수 (D열)
    const lossesB = b[3]; // 패배수 (D열)

    // '베타시즌'은 가장 마지막에 오도록 처리합니다.
    if (seasonA === '베타시즌' && seasonB !== '베타시즌') {
      return 1;
    }
    if (seasonA !== '베타시즌' && seasonB === '베타시즌') {
      return -1;
    }

    // 시즌을 숫자로 변환하여 비교합니다.
    const seasonNumA = parseInt(seasonA.replace('시즌', ''), 10);
    const seasonNumB = parseInt(seasonB.replace('시즌', ''), 10);

    // 1. 시즌(숫자) 내림차순 정렬
    if (seasonNumA !== seasonNumB) {
      return seasonNumB - seasonNumA;
    }

    // 2. 승리수 내림차순 정렬
    if (winsA !== winsB) {
      return winsB - winsA;
    }

    // 3. 패배수 오름차순 정렬
    return lossesA - lossesB;
  });

  // 정렬된 데이터를 시트에 다시 씁니다.
  dataRange.setValues(data);
}

/**
 * 배열에서 가능한 모든 조합(Combination)을 생성합니다. (itertools.combinations 대체)
 * @param {Array<string>} array - 조합을 생성할 배열.
 * @param {number} size - 조합을 만들 요소의 개수 (듀오이므로 2).
 * @returns {Array<Array<string>>} 조합 배열.
 */
function combinations(array, size) {
  const result = [];
  function generate(start, current) {
    if (current.length === size) {
      result.push([...current]);
      return;
    }
    for (let i = start; i < array.length; i++) {
      current.push(array[i]);
      generate(i + 1, current);
      current.pop();
    }
  }
  generate(0, []);
  return result;
}

/**
 * 두 이름의 순서를 정렬하여 듀오 조합명을 생성합니다.
 * @param {string} name1 
 * @param {string} name2 
 * @returns {string} 예: "Alice&Bob"
 */
function createDuoName(name1, name2) {
  if (!name1 || !name2) return null; 
  const names = [name1, name2].sort();
  return `${names[0]}&${names[1]}`;
}

function cancelDuoRecordsTransaction() {
  
  // ----------------------------------------------------
  // 1. 초기 설정 및 시즌 정보 가져오기
  // ----------------------------------------------------
  
  const currentSeason = getCurrentSeason(); // 시즌 정보 가져오기
    
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  
  if (!spreadsheet) {
    Logger.log(`오류: "${spreadsheetName}" 스프레드시트를 찾을 수 없습니다.`);
    SpreadsheetApp.getUi().alert(`오류: 스프레드시트를 찾을 수 없습니다.`);
    return;
  }
  
  const duosheet = spreadsheet.getSheetByName("듀오전적");
  const activeSheet = spreadsheet.getSheetByName("디스코드 연동");

  if (!duosheet || !activeSheet) {
    Logger.log("오류: 필요한 시트를 찾을 수 없습니다.");
    SpreadsheetApp.getUi().alert("오류: 시트 이름을 확인해 주세요.");
    return;
  }

  // ----------------------------------------------------
  // 2. 캐릭터명 파싱 로직 (A10, A12 셀)
  // ----------------------------------------------------

  const winTeamString = activeSheet.getRange("A10").getValue().toString().trim();
  const loseTeamString = activeSheet.getRange("A12").getValue().toString().trim();

  if (!winTeamString || !loseTeamString) {
    Logger.log("오류: A10 또는 A12 셀에 데이터가 없습니다. 파싱 취소.");
    SpreadsheetApp.getUi().alert("오류: A10/A12 셀을 확인해주세요.");
    return;
  }

  const winTeamNames = winTeamString.split('/');
  const loseTeamNames = loseTeamString.split('/');
  
  const druWinName = winTeamNames[0] ? winTeamNames[0].trim() : '';
  const assWinName = winTeamNames[1] ? winTeamNames[1].trim() : '';
  const necWinName = winTeamNames[2] ? winTeamNames[2].trim() : '';
  const smtWinName = winTeamNames[3] ? winTeamNames[3].trim() : '';
  
  const druLoseName = loseTeamNames[0] ? loseTeamNames[0].trim() : '';
  const assLoseName = loseTeamNames[1] ? loseTeamNames[1].trim() : '';
  const necLoseName = loseTeamNames[2] ? loseTeamNames[2].trim() : '';
  const smtLoseName = loseTeamNames[3] ? loseTeamNames[3].trim() : '';
  
  // ----------------------------------------------------
  // 3. 듀오 조합 생성 및 전적 취소 로직
  // ----------------------------------------------------
  
  const winTeamMembers = [druWinName, assWinName, necWinName, smtWinName].filter(name => name);
  const loseTeamMembers = [druLoseName, assLoseName, necLoseName, smtLoseName].filter(name => name);
  
  const winDuos = combinations(winTeamMembers, 2).map(pair => createDuoName(pair[0], pair[1]));
  const loseDuos = combinations(loseTeamMembers, 2).map(pair => createDuoName(pair[0], pair[1]));
  
  // 듀오 전적 취소 실행 (시즌 정보 전달)

  processDuoCancellation(duosheet, winDuos, 3, "승리", currentSeason); 
  processDuoCancellation(duosheet, loseDuos, 4, "패배", currentSeason);

  activeSheet.getRange("A12").setValue("");

  Logger.log('🎉 듀오 전적 취소 처리가 최종 완료되었습니다.');
}

/**
 * 듀오 조합 리스트를 순회하며 전적을 업데이트합니다.
 * @param {GoogleAppsScript.Spreadsheet.Sheet} sheet - 듀오 전적 시트.
 * @param {Array<string>} duoList - 업데이트할 듀오 조합 목록.
 * @param {number} columnToUpdate - 업데이트할 열 번호 (3: 승리(C), 4: 패배(D)).
 * @param {string} status - 로그 메시지에 사용할 상태 ("승리" 또는 "패배").
 * @param {string} season - 현재 시즌 정보 (A열과 비교할 값).
 */
function processDuoCancellation(sheet, duoList, columnToUpdate, status, season) {
  let updatedCount = 0;
  
  // B열 전체에서 검색 (듀오 조합명이 있는 열)
  const searchRange = sheet.getRange('B:B'); 
  
  for (const duo of duoList) {
    if (!duo) continue; 
    
    // 듀오 조합명으로 B열에서 해당 행 찾기
    const foundCell = searchRange.createTextFinder(duo)
                                 .matchEntireCell(true)
                                 .findNext();
    
    if (foundCell) {
      const duoRow = foundCell.getRow();
      
      // *** 시즌 체크 로직 추가 ***
      // A열 (1열)의 값을 가져와 현재 시즌과 비교합니다.
      const seasonCell = sheet.getRange(duoRow, 1); // 1열: A열
      const rowSeason = seasonCell.getValue().toString().trim();
      
      if (rowSeason !== season) {
        Logger.log(`정보: "${duo}" 조합을 찾았으나, 시즌(${rowSeason})이 현재 시즌(${season})과 일치하지 않아 건너뜁니다.`);
        continue; // 시즌 불일치 시 다음 듀오로 넘어감
      }
      // *************************

      try {
        // 현재 승/패 횟수 가져오기
        const targetRange = sheet.getRange(duoRow, columnToUpdate);
        const currentValue = parseInt(targetRange.getValue());
        
        if (!isNaN(currentValue) && currentValue > 0) {
          // 값 -1로 업데이트
          targetRange.setValue(currentValue - 1);
          updatedCount++;
        } else {
          Logger.log(`정보: 듀오 "${duo}" (시즌: ${season})의 현재 ${status} 횟수가 0이하여서 취소하지 않았습니다.`);
        }
      } catch (e) {
        Logger.log(`경고: 듀오 전적 취소 중 오류 발생 - ${duuo}, ${e}`);
      }
    } else {
      Logger.log(`경고: 듀오 전적 시트에서 "${duo}" 조합을 찾을 수 없습니다.`);
    }
  }
  
  Logger.log(`총 ${updatedCount}건의 ${status} 듀오 전적 (시즌: ${season})이 취소 업데이트 되었습니다.`);
}

function updateDuoStats() {

  const duoSheet = ss.getSheetByName('듀오전적');

  if (!historySheet) {
    SpreadsheetApp.getUi().alert('\'전적이력\' 시트를 찾을 수 없습니다.');
    return;
  }
  
  // 'duotemp' 오류 메시지를 '듀오전적' 시트를 찾지 못했다는 메시지로 수정했습니다.
  if (!duoSheet) {
    SpreadsheetApp.getUi().alert('\'듀오전적\' 시트를 찾을 수 없습니다.');
    return;
  }

  const historyData = historySheet.getDataRange().getValues();
  const duoStats = {};

  // 헤더를 제외하고 데이터를 순회합니다. (1행부터 시작)
  for (let i = 1; i < historyData.length; i++) {
    const row = historyData[i];
    // H열(인덱스 7)의 시즌 정보를 가져옵니다.
    const season = row[7];
    const winningTeamString = row[2]; // C열
    const losingTeamString = row[4]; // E열

    // <br>과 괄호 안의 내용을 모두 제거하고 닉네임만 추출하는 함수
    const extractPlayers = (teamString) => {
      if (!teamString) return [];
      const cleanedString = teamString.replace(/\(.*?\)/g, '').replace(/<br>/g, '').trim();
      return cleanedString.split(/\s+/).filter(Boolean);
    };

    const winningPlayers = extractPlayers(winningTeamString);
    const losingPlayers = extractPlayers(losingTeamString);

    // 승리팀 듀오 조합 처리
    // generateDuos() 함수는 외부 또는 전역으로 정의되어 있다고 가정합니다.
    const winningDuos = generateDuos(winningPlayers);
    winningDuos.forEach(duo => {
      const duoKey = `${duo[0]}&${duo[1]}`;
      if (!duoStats[duoKey]) {
        duoStats[duoKey] = {};
      }
      if (!duoStats[duoKey][season]) {
        duoStats[duoKey][season] = { wins: 0, losses: 0 };
      }
      duoStats[duoKey][season].wins++;
    });

    // 패배팀 듀오 조합 처리
    const losingDuos = generateDuos(losingPlayers);
    losingDuos.forEach(duo => {
      const duoKey = `${duo[0]}&${duo[1]}`;
      if (!duoStats[duoKey]) {
        duoStats[duoKey] = {};
      }
      if (!duoStats[duoKey][season]) {
        duoStats[duoKey][season] = { wins: 0, losses: 0 };
      }
      duoStats[duoKey][season].losses++;
    });
  }

  // 결과를 배열에 담기
  let output = [];
  for (const duoKey in duoStats) {
    for (const season in duoStats[duoKey]) {
      const stats = duoStats[duoKey][season];
      // 승률 수식 삽입을 위해 빈 문자열을 추가하여 5열을 만듭니다.
      output.push([season, duoKey, stats.wins, stats.losses, '']); 
    }
  }

  // 시즌(숫자) -> 승리 순으로 정렬
  output.sort((a, b) => {
    // 시즌 값을 숫자로 변환 (예: '시즌13' -> 13)
    const seasonA = parseInt(a[0].replace('시즌', ''), 10);
    const seasonB = parseInt(b[0].replace('시즌', ''), 10);

    // 1. 시즌이 높은 순으로 정렬 (내림차순)
    if (seasonB !== seasonA) {
      return seasonB - seasonA;
    }
    // 2. 같은 시즌이면 승리 횟수가 높은 순으로 정렬 (내림차순)
    return b[2] - a[2];
  });
  
  // 헤더를 포함한 최종 데이터 생성
  // 승률 헤더 '승률'을 추가합니다.
  const finalOutput = [['시즌', '듀오조합', '승', '패', '승률'], ...output];

  // 기존 데이터 삭제 후 새 데이터 쓰기
  duoSheet.clear();
  if (finalOutput.length > 1) {
    // 1. 집계된 데이터를 시트에 씁니다. (5열 포함)
    const numCols = finalOutput[0].length; // 5
    duoSheet.getRange(1, 1, finalOutput.length, numCols).setValues(finalOutput);

// 4명의 플레이어로부터 모든 듀오 조합을 생성하는 헬퍼 함수
function generateDuos(players) {
  const duos = [];
  if (players.length !== 4) return duos;

  for (let i = 0; i < players.length - 1; i++) {
    for (let j = i + 1; j < players.length; j++) {
      const p1 = players[i];
      const p2 = players[j];
      const sortedDuo = [p1, p2].sort();
      duos.push(sortedDuo);
    }
  }
  return duos;
}


/**
 * 현재 활성화된 스프레드시트의 B열("듀오조합") 데이터를 정렬하여 일관성을 확보합니다.
 * 예: "에겐남&원" -> "원&에겐남" (또는 그 반대)
 */
function sortDuoCombinations() {
  // 현재 활성화된 스프레드시트를 가져옵니다.
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = spreadsheet.getSheetByName('듀오전적');
  
  // B열의 데이터가 시작되는 행 (일반적으로 헤더 다음인 2행)과 마지막 행을 계산합니다.
  // getRange(시작 행, 시작 열, 행 개수, 열 개수)
  const lastRow = sheet.getLastRow();
  
  // 데이터가 2행부터 시작해야 하므로, 마지막 행이 1행(헤더만 있음)보다 작으면 작업을 중단합니다.
  if (lastRow < 2) {
    Logger.log("헤더를 제외하고 처리할 데이터가 없습니다 (2행 이상 필요).");
    SpreadsheetApp.getUi().alert("처리할 데이터가 2행 이상 없습니다.");
    return;
  }

  // B열의 2행부터 마지막 행까지의 범위를 가져옵니다. (B2:B[lastRow])
  // numRows는 lastRow - 1 입니다. (예: 10행이 마지막이면 2~10행이므로 9개)
  const numRows = lastRow - 1;
  const range = sheet.getRange(2, 2, numRows, 1);
  const values = range.getValues(); // 2차원 배열 형태로 데이터를 가져옵니다.
  
  // 정렬된 결과를 저장할 새 배열을 준비합니다.
  let sortedValues = [];

  // 각 셀의 값을 순회하며 정렬 작업을 수행합니다.
  for (let i = 0; i < values.length; i++) {
    const duoString = values[i][0]; // 현재 셀의 값 (예: "유즈&에겐남")
    
    if (typeof duoString === 'string' && duoString.includes('&')) {
      // 1. '&' 기호를 기준으로 문자열을 나눕니다.
      let names = duoString.split('&');
      
      // 2. 각 이름을 앞뒤 공백 없이 다듬습니다. (선택 사항이지만 안전을 위해)
      names = names.map(name => name.trim());
      
      // 3. 닉네임 목록을 정렬합니다. (가나다순 / 알파벳순)
      names.sort();
      
      // 4. 정렬된 닉네임을 다시 '&' 기호로 연결합니다.
      const sortedDuoString = names.join('&');
      
      // 5. 정렬된 문자열을 새 배열에 저장합니다. (2차원 배열 형태로)
      sortedValues.push([sortedDuoString]);
    } else {
      // '&' 기호가 없거나 문자열이 아닌 경우, 원래 값을 그대로 유지합니다.
      sortedValues.push([duoString]);
    }
  }

  // 정렬된 전체 데이터를 B열에 다시 씁니다.
  range.setValues(sortedValues);
  
  Logger.log("B열 듀오 조합 정렬 완료.");
  SpreadsheetApp.getUi().alert("B열 듀오 조합 정렬을 완료했습니다!");
}
